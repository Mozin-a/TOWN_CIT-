function render(items,title){
 document.getElementById("home").classList.add("hidden");
 document.getElementById("results").classList.remove("hidden");
 document.getElementById("resultTitle").textContent=title;
 let area=document.getElementById("laws");
 area.innerHTML="";
 if(!items.length){
  area.innerHTML="<p>ماكو نتائج مطابقة للبحث.</p>";
  return;
 }
 items.forEach(l=>{
  let el=document.createElement("div");
  el.className="law";
  el.innerHTML=`<strong>المادة (${l.n})</strong><br>${l.text}<br><small>${l.chapter}</small>`;
  area.appendChild(el);
 });
 window.scrollTo(0,0);
}

});

 render(allLaws.filter(l=>l.n>=c[1]&&l.n<=c[2]),c[0]);
